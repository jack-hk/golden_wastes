{{- $result := dict "editable" false "url" "" -}} {{- $page := . -}} {{- with $page.File -}} {{- $path := replace .Path "\\" "/" -}} {{- $schema := "" -}} {{- $schemaType := "collection" -}} {{- if hasPrefix $path "article/" -}} {{- $schema = "articles" -}} {{- else if hasPrefix $path "campaign/" -}} {{- $schema = "campaigns" -}} {{- else if hasPrefix $path "monster/" -}} {{- $schema = "monsters" -}} {{- else if hasPrefix $path "location/" -}} {{- $schema = "locations" -}} {{- else if hasPrefix $path "class/" -}} {{- $schema = "classes" -}} {{- else if hasPrefix $path "categories/" -}} {{- $schema = "category_portals" -}} {{- else if eq $path "_index.md" -}} {{- $schema = "home" -}} {{- $schemaType = "file" -}} {{- end -}} {{- if $schema -}} {{- $cms := site.Params.pagesCMS -}} {{- $baseURL := strings.TrimRight "/" ($cms.baseURL | default "https://app.pagescms.org") -}} {{- $owner := $cms.owner | default "" -}} {{- $repo := $cms.repo | default "" -}} {{- $branch := $cms.branch | default "main" -}} {{- if and $owner $repo -}} {{- $url := "" -}} {{- $encodedBranch := replace ($branch | urlquery) "+" "%20" -}} {{- $encodedSchema := replace ($schema | urlquery) "+" "%20" -}} {{- if eq $schemaType "file" -}} {{- $url = printf "%s/%s/%s/%s/file/%s" $baseURL $owner $repo $encodedBranch $encodedSchema -}} {{- else -}} {{- $repoPath := printf "content/%s" $path -}} {{- $encodedRepoPath := replace ($repoPath | urlquery) "+" "%20" -}} {{- $url = printf "%s/%s/%s/%s/collection/%s/edit/%s" $baseURL $owner $repo $encodedBranch $encodedSchema $encodedRepoPath -}} {{- end -}} {{- $result = dict "editable" true "url" $url -}} {{- end -}} {{- end -}} {{- end -}} {{- return $result -}}
