" \t\r\n" -}} {{- if $view -}} {{- if not (in $views $view) -}} {{- errorf "The responsive shortcode %s option contains %q; supported views are mobile, tablet, and desktop: %s" $mode $view $.Position -}} {{- else if not (in $selectedViews $view) -}} {{- $selectedViews = $selectedViews | append $view -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $selectedViews -}} {{- errorf "The responsive shortcode %s option must contain at least one view: %s" $mode .Position -}} {{- else if gt (len $selectedViews) 3 -}} {{- errorf "The responsive shortcode supports no more than three views: %s" .Position -}} {{- end -}} {{- $classes := slice "responsive-content" (printf "responsive-content--%s" $mode) -}} {{- range $selectedViews -}} {{- $classes = $classes | append (printf "responsive-content--%s-%s" $mode .) -}} {{- end -}}
